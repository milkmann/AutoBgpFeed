Примеры подключения для любых роутеров и ОС
Выберите вашу платформу для получения готового скрипта настройки BGP-клиента:
# Выполните в терминале MikroTik RouterOS 7 (замените wireguard1 на имя вашего VPN-интерфейса):
# 1. Создаем правило фильтрации входящих BGP-маршрутов
/routing/filter/rule add chain=antifilter-in disabled=no rule="if (bgp-communities includes 65000:1000) { set gw wireguard1; accept; } else { reject; }"

# 2. Подключаемся к BGP-узлу AutoBgpFeed
/routing/bgp/connection add name="bgp-feed" remote.address={{ server_ip }} remote.port=179 remote.as={{ server_asn }} \
    as=64999 local.role=ebgp multihop=yes connect=yes listen=no \
    routing-table=main input.filter=antifilter-in output.filter-chain=discard
# Выполните в терминале MikroTik RouterOS 6:
# 1. Создаем фильтры маршрутизации
/routing filter add chain=antifilter-in bgp-communities=65000:1000 set-gateway=wireguard1 action=accept
/routing filter add chain=antifilter-in action=reject

# 2. Настраиваем BGP Peer
/routing bgp instance set default as=64999
/routing bgp peer add name="bgp-feed" remote-address={{ server_ip }} remote-as={{ server_asn }} \
    multihop=yes in-filter=antifilter-in out-filter=discard hold-time=3m
# Для Keenetic через Entware (opkg update && opkg install bird2): добавьте в /opt/etc/bird.conf:
protocol bgp bgp_feed {
    local as 64999;
    neighbor {{ server_ip }} as {{ server_asn }};
    multihop;
    ipv4 {
        import filter {
            if (65000, 1000) ~ bgp_community then {
                gw = 10.0.0.1; # IP вашего VPN-шлюза (WireGuard/OpenVPN)
                accept;
            }
            reject;
        };
        export none;
    };
}
# Для OpenWrt (opkg install bird2) или Linux-клиента в /etc/bird.conf:
protocol bgp bgp_feed {
    local as 64999;
    neighbor {{ server_ip }} as {{ server_asn }};
    multihop;
    ipv4 {
        import filter {
            if (65000, 1000) ~ bgp_community then {
                gw = 10.0.0.1; # IP адрес VPN-шлюза (wg0/tun0)
                accept;
            }
            reject;
        };
        export none;
    };
}
# Выполните в CLI VyOS / Ubiquiti EdgeRouter:
# 1. Настройка политик фильтрации
set policy route-map BGP-OUT-DISCARD rule 10 action reject
set policy route-map BGP-IN-VPN rule 10 match community 65000:1000
set policy route-map BGP-IN-VPN rule 10 set next-hop 10.0.0.1
set policy route-map BGP-IN-VPN rule 20 action reject

# 2. Настройка BGP пира
set protocols bgp system-as 64999
set protocols bgp neighbor {{ server_ip }} remote-as {{ server_asn }}
set protocols bgp neighbor {{ server_ip }} ebgp-multihop 10
set protocols bgp neighbor {{ server_ip }} address-family ipv4-unicast route-map import BGP-IN-VPN
set protocols bgp neighbor {{ server_ip }} address-family ipv4-unicast route-map export BGP-OUT-DISCARD
# Конфигурация для Cisco IOS / IOS-XE:
! 1. Фильтры Community и Route-Map
ip community-list standard CL-FEED permit 65000:1000
!
route-map BGP-IN-VPN permit 10
 match community CL-FEED
 set ip next-hop 10.0.0.1
route-map BGP-OUT-DISCARD deny 10
!
! 2. Настройка BGP
router bgp 64999
 neighbor {{ server_ip }} remote-as {{ server_asn }}
 neighbor {{ server_ip }} ebgp-multihop 10
 address-family ipv4
  neighbor {{ server_ip }} activate
  neighbor {{ server_ip }} route-map BGP-IN-VPN in
  neighbor {{ server_ip }} route-map BGP-OUT-DISCARD out
 exit-address-family
