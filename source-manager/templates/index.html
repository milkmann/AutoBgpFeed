Конфигурация BGP на роутере (с защитным фильтром)
Рекомендуемые безопасные правила (с защитой от 0.0.0.0/0, RFC1918 и поддержкой нескольких VPN через BGP communities):
# 1. Защита от опасных/приватных маршрутов + распределение по VPN-туннелям (RouterOS 7)
/routing/filter/rule
add chain=antifilter-in disabled=no rule="if (dst == 0.0.0.0/0 || dst-len < 8 || dst in 10.0.0.0/8 || dst in 172.16.0.0/12 || dst in 192.168.0.0/16 || dst in 100.64.0.0/10 || dst in {{ server_ip }}/32) { reject; }"
{% for vp in vpn_profiles if vp.community != (server_asn|string ~ ':1000') %}add chain=antifilter-in disabled=no rule="if (bgp-communities includes {{ vp.community }}) { set gw {{ vp.name }}; accept; }"
{% endfor %}add chain=antifilter-in disabled=no rule="if (bgp-communities includes {{ server_asn }}:1000) { set gw wireguard1; accept; } else { reject; }"
add chain=discard disabled=no rule="reject;"

# 2. Подключение к AutoBgpFeed (для RouterOS 7.20+ предварительно создайте /routing/bgp/instance)
/routing/bgp/connection
add name="bgp-feed" remote.address={{ server_ip }} remote.port=179 remote.as={{ server_asn }} \
    as=64999 local.role=ebgp multihop=yes connect=yes listen=no \
    routing-table=main input.filter=antifilter-in input.limit-process-routes-ipv4=25000 output.filter-chain=discard
/routing filter add chain=antifilter-in prefix=0.0.0.0/0 action=reject
/routing filter add chain=antifilter-in bgp-communities={{ server_asn }}:1000 set-gateway=wireguard1 action=accept
/routing filter add chain=antifilter-in action=reject
/routing bgp instance set default as=64999
/routing bgp peer add name="bgp-feed" remote-address={{ server_ip }} remote-as={{ server_asn }} multihop=yes in-filter=antifilter-in out-filter=discard hold-time=3m
protocol bgp bgp_feed {
    local as 64999;
    neighbor {{ server_ip }} as {{ server_asn }};
    multihop;
    ipv4 {
        import filter {
            if net = 0.0.0.0/0 || net.len < 8 then reject;
            if ({{ server_asn }}, 1000) ~ bgp_community then { gw = 10.0.0.1; accept; }
            reject;
        };
        export none;
    };
}
protocol bgp bgp_feed {
    local as 64999;
    neighbor {{ server_ip }} as {{ server_asn }};
    multihop;
    ipv4 {
        import filter {

            if ({{ server_asn }}, 1000) ~ bgp_community then { gw = 10.0.0.1; accept; }
            reject;
        };
        export none;
    };
}
